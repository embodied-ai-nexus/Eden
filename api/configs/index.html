    Configs and Task Registry¶

    A task is an EdenConfig subclass: one validated Pydantic object that holds every option group an environment is built from. This page documents the config hierarchy, its serialization helpers, and the task registry that finds configs by name. For how to write a task config, read the task configuration guide; for variants and their CLI flags, read the TaskMods guide.

    from eden.tasks import TASK_REGISTRY, TaskMod
from eden.utils.configs import EdenConfig, EdenMimicConfig, EdenRLConfig

    Each config class pairs with exactly one environment class (EdenConfig with EnvBase, EdenRLConfig with RLEnvBase, EdenMimicConfig with MimicRLEnvBase); a mismatch raises TypeError at construction. See Environments.

    The camera a recording of this task renders, added as rec by :meth:add_recording_camera.

    eden inference --rec adds it; so does any script that records the task and calls :meth:add_recording_camera. A method rather than a cameras_options entry so that only a recording run builds it: a camera in the training config costs every training run its setup (and whatever its overlays allocate) though nothing renders it, and a checkpoint replays the config it was trained with.     def add_recording_camera(self) -> None:
    """Add :meth:`recording_camera` to this config as its ``rec`` camera, in place, unless it already has one.

    A ``rec`` camera the config already has (a ``cameras_options: {rec: {...}}`` override) is kept: replacing it
    would silently discard the override. It replaces the task's camera whole, overlays included, so the overlays
    it lacks are named in a warning.
    """
    if self.cameras_options is None:
        self.cameras_options = CamerasOptions()
    declared = getattr(self.cameras_options, "rec", None)
    if declared is None:
        self.cameras_options.rec = self.recording_camera()
        return
    # By value: an overlay read back from a saved config is equal to the one written (its parameters are given their
    # declared types again, so a tuple is not a list).
    dropped = [overlay for overlay in self.recording_camera().overlays if overlay not in declared.overlays]
    if dropped:
        en.logger.warning(
            f"The config declares its own `rec` camera, which replaces {type(self).__name__}.recording_camera() "
            f"whole, so the recording leaves out that camera's overlays {[overlay.name for overlay in dropped]} "
            "(task state the scene does not render, such as ink). To keep them, add them to "
            "cameras_options.rec.overlays, e.g. the entries of config.recording_camera().overlays."
        )

    def save_as_file(self, path: str, lock_config: bool = True):
    """
    Save the config as either JSON or YAML file with metadata envelope.

    Parameters
    ----------
    path: str
        The path to save the config.
    lock_config: bool
        Whether to lock the config after saving to prevent further modifications.
    """
    file_type = path.split(".")[-1].lower()
    if file_type not in ["json", "yaml", "yml"]:
        raise ValueError(f"Unsupported config file extension: {file_type}. Use .json, .yaml, or .yml")

    result = self._serialize()

    if lock_config:
        self.lock()

    dirpath = os.path.dirname(path)
    if dirpath:
        os.makedirs(dirpath, exist_ok=True)
    with open(path, "w") as f:
        if file_type == "json":
            json.dump(result, f, indent=4)
        else:
            yaml.safe_dump(result, f, sort_keys=False)

    def save_as_json(self, path: str | None = None, lock_config: bool = True):
    """
    Save the config as a JSON file with metadata envelope.

    Parameters
    ----------
    path: str | None
        The path to save the config. If not provided, the config will be saved to the log directory.
    lock_config: bool
        Whether to lock the config after saving to prevent further modifications.

    Raises
    ------
    ValueError
        If ``path`` does not end in ``.json`` (case-insensitive). Nothing is written and the
        config is left unlocked.
    """
    path = path or f"{en.log_dir}/config.json"
    # Not an ``assert``: ``python -O`` strips those, and ``save_as_file`` dispatches on the
    # suffix — so ``save_as_json("cfg.txt")`` would silently write a file whose extension
    # ``load_from_file`` refuses. Matched case-insensitively, as ``save_as_file`` is.
    if not path.lower().endswith(".json"):
        raise ValueError(
            f"save_as_json expects a path ending in '.json', got '{path}'. "
            f"Use save_as_yaml for YAML, or save_as_file to dispatch on the extension."
        )
    self.save_as_file(path, lock_config)

    load_from_file classmethod ¶

    load_from_file(path: str) -> EdenConfig

    Load the config from a JSON or YAML file, reconstructing its saved class.

    The class the file was saved from is recorded in its metadata, so EdenConfig.load_from_file("logs/.../config.json") returns the concrete task config (ReacherConfig) rather than a bare EdenConfig — which is what makes a generic loader like eden train --config <a saved config.json> work at all: a base EdenConfig has nowhere to put reward_options and rejects the file outright.

    """Reconstruct a saved config from already-read data — see :meth:`load_from_file`.

    Split out so a caller that has to *inspect* the data before deciding what to do with it
    (``eden train --config``, which reads :func:`is_full_config` to tell a full save from a
    partial override) can reconstruct without reading the file a second time.

    Parameters
    ----------
    data: dict
        The file's contents, in either the ``_meta``/``config`` envelope or the legacy flat
        layout. Consumed destructively — pass a copy to keep it.
    source: str
        Where ``data`` came from, for error messages.

    Returns
    -------
    EdenConfig
        An instance of the saved class (a ``cls`` instance when the data records no class).
    """
    meta = data.pop("_meta", None)
    if meta is not None and "config" in data:
        # Envelope format: {"_meta": {...}, "config": {...}}
        data = data["config"]
    else:
        # Legacy flat format: metadata sat alongside the fields. It carries the same
        # class tags, so it resolves the saved class the same way.
        meta = {key: data.pop(key, None) for key in (*OPTION_METADATA_KEYS, "eden_version", "genesis_version")}

    options = to_options(data, cls._saved_config_class(meta, source))

    for key, name, current in (
        ("eden", "Eden", get_eden_provenance()),
        ("genesis", "Genesis", get_package_provenance("genesis-world")),
    ):
        if change := describe_provenance_change(meta.get(key), current):
            en.logger.warning(f"{name} differs from the one that saved {source}: {change}.")

    return options

    Reading and converting config files¶

    Eden config hierarchy (EdenConfig/EdenRLConfig/EdenMimicConfig) and serialization.

    • :class:EdenConfig — base config for all environments (EnvBase).
    • :class:EdenRLConfig — adds reward, termination, command, curriculum, and runner options.
    • :class:EdenMimicConfig — adds motion options for motion-imitation tasks.

    Root config classes use Pydantic extra="forbid" (so misspelled fields raise), while inner ManagerOptions / SceneOptions keep extra="allow" to hold dynamic term/entity children. Saved configs use a _meta/config envelope; nested Options recover their type from _option_module_ / _option_class_ keys.

    The root config recovers its type the same way: :meth:EdenConfig.load_from_file reads the saved class out of _meta and reconstructs that class, so a generic EdenConfig.load_from_file(path) returns the concrete task config it was saved from. The requested class still constrains the result — the saved class must be a subclass of it — and files with no class metadata load as the requested class. Round-tripping a config through save/load is therefore type-preserving and idempotent (pinned by tests/utils/test_configs.py::test_every_task_config_round_trips_to_a_fixed_point).

    def serialize_obj_with_metadata(obj):
    """Recursively serialize an object, capturing Options type metadata.

    Captures ``_option_module_`` and ``_option_class_`` metadata from Options
    instances. This works by using ``model_dump()`` for data but accessing actual
    Option objects to extract their metadata.

    NamedTuple values (e.g. Genesis ``TemperatureProperties``) subclass ``tuple`` and are
    serialized as plain lists of fields in field order; ``to_options`` reconstructs them
    using the surrounding Pydantic field annotations on load.
    """
    if isinstance(obj, Options):
        # A field left at a None default is dumped as absent, not as an explicit None: loading would
        # otherwise mark it set, and Genesis reads set-ness — a solver whose options merely list `dt`
        # among `model_fields_set` divides the sim dt by it (`MochiOptions(dt=None)` would crash the build).
        unset_none = {name for name in type(obj).model_fields if name not in obj.model_fields_set}
        unset_none = {name for name in unset_none if getattr(obj, name, None) is None}
        result: dict = obj.model_dump(exclude=unset_none)

        # Stamp type metadata at dump time. The getattr fallback covers extra="forbid" options
        # (which don't carry it in __pydantic_extra__), so entities stored as unannotated scene
        # children still recover their concrete class on load even though they forbid extras.
        result["_option_module_"] = getattr(obj, "_option_module_", obj.__module__)
        result["_option_class_"] = getattr(obj, "_option_class_", obj.__class__.__name__)

        # Now recursively process each field value to capture nested Options metadata
        for field_name in list(result.keys()):
            if field_name not in OPTION_METADATA_KEYS:
                # Try to get the actual object using getattr to handle all cases
                # (including fields not in __dict__ due to defaults, properties, etc)
                try:
                    actual_value = getattr(obj, field_name)
                    # If the actual value is an Options object, recursively serialize it
                    if isinstance(actual_value, Options):
                        result[field_name] = serialize_obj_with_metadata(actual_value)
                    else:
                        # Recurse over the live field value so container fields
                        # (for example list[Union[...Options]]) keep subclass data
                        # that may be erased by model_dump() schema narrowing.
                        result[field_name] = serialize_obj_with_metadata(actual_value)
                except AttributeError:
                    # Field doesn't exist, keep the dumped value and recurse
                    result[field_name] = serialize_obj_with_metadata(result[field_name])

        return result
    elif isinstance(obj, BaseModel):
        # For other BaseModel instances, serialize and recurse
        result = obj.model_dump()
        for key, value in list(result.items()):
            result[key] = serialize_obj_with_metadata(value)
        return result
    elif isinstance(obj, dict):
        return {key: serialize_obj_with_metadata(value) for key, value in obj.items()}
    elif isinstance(obj, (list, tuple)):
        return [serialize_obj_with_metadata(item) for item in obj]
    elif isinstance(obj, Enum):
        return obj.value
    elif isinstance(obj, np.ndarray):
        return obj.tolist()
    elif isinstance(obj, torch.Tensor):
        return obj.detach().cpu().numpy().tolist()

    Task registry¶

    Tasks register themselves with @TASK_REGISTRY.register() on the config class. The registry scans the source of eden.tasks (and of every package an installed distribution names under the eden.tasks entry-point group) on first lookup without importing it, so there is no central index to update. TASK_REGISTRY.build(name, **kwargs) returns a configured instance with the task's TaskMods applied, which is what eden train does; TASK_REGISTRY.get(name) returns the bare class, and instantiating that yourself skips every mod.

    TaskRegistry ¶

    TaskRegistry()

    Task registry with two lookup paths, tried in order.

    1. Eager registry — populated by @TASK_REGISTRY.register() decorators or explicit register(cls, ...) calls.
    2. Auto-discovery — an AST scan of the configured search packages (default ["eden.tasks"]) finds @TASK_REGISTRY.register() decorators in every *.py file without importing them. On first :meth:get miss we import only the one file that matches the requested name; that triggers the decorator and populates the eager registry.

    Tasks self-register at the definition site — no central index in eden/tasks/__init__.py to keep in sync. Out-of-tree tasks join the scan through the :data:TASK_ENTRY_POINT_GROUP entry point of the distribution that ships them, so eden list / eden train --task see them without importing anything; a script can also call :meth:add_search_path before its first lookup.

    Auto-discovery rules (files that are NOT scanned):

    • __init__.py is skipped — put decorated classes in a sibling module such as config.py.
    • files whose name starts with _ are skipped (e.g. _meta.py) — private modules are treated as implementation details.

    Auto-discovery rules (decorator recognition):

    • The scan recognizes @<name>.register where <name> is bound, in the same file, to TASK_REGISTRY via an absolute import from eden.tasks or eden.tasks.registry (from ... import TASK_REGISTRY or ... as X). Both forms work.
    • Using the literal name TASK_REGISTRY without importing it from a recognized location emits a warning — the scan will not silently miss the registration.

        # keyword-only init param on a registered mod — that's the most

        # never reaches the CLI surface). Match on the prefixed flag form
        # so the hint also fires for prefixed mods.
        kwonly_index: dict[str, tuple[TaskMod, str]] = {}
        for mod in mods:
            for field in _keyword_only_fields(type(mod)):
                kwonly_index[mod.flag_for(field)] = (mod, field)
        kwonly_hits: list[str] = []

                mod, field = kwonly_index[u]
                kwonly_hits.append(
                    f"'{u}' is keyword-only on {type(mod).__name__} (as '{field}') — "
                    f"keyword-only params are non-CLI ctor state. Set it at registration "
                    f"with modifiers=({type(mod).__name__}({field}=...),); if you own "

            f"Unknown modifier kwargs for task '{name}': {sorted(unknown)}. "
            f"Known: {sorted(flag_map) or 'none'}{hint}"
        )

    # Group kwargs per mod, stripping the prefix and validating each value
    # against its declared annotation (Literal membership + primitive type).
    updates_per_mod: list[dict[str, Any]] = [{} for _ in mods]
    for flag, value in modifier_kwargs.items():
        i, field_name, annotation = flag_map[flag]
        _check_cli_value(field_name, annotation, value)
        updates_per_mod[i][field_name] = value

    for mod, updates in zip(mods, updates_per_mod):
        if updates:
            # Deep copy so per-build overrides don't mutate the registered
            # instance; deep so mutable instance state (lists/dicts a
            # subclass might keep) is not aliased across builds.
            applied = copy.deepcopy(mod)
            for field_name, value in updates.items():
                setattr(applied, field_name, value)
        else:
            applied = mod
        config = applied.apply(config)
    return config

    TaskMod ¶

    Tool that modifies an EdenConfig instance.

    TaskMod is a plain Python class (not a Pydantic model) because it is logic with a few parameters, not user-facing data. EdenConfig is the Pydantic data model; TaskMod operates on it.

    To define one: subclass, declare CLI-tunable parameters as positional-or-keyword __init__ arguments with type annotations, and implement :meth:apply. Non-CLI constructor state goes in keyword-only parameters (after *) — the CLI introspection skips those.

    Quick reference (this is the convention agents miss most often)::

    def __init__(
    self,
    difficulty: Literal["easy", "hard"] = "easy",  # ← CLI flag (--difficulty)
    *,
    entity_name: str = "robot",                    # ← non-CLI ctor state
):
    self.difficulty = difficulty
    self.entity_name = entity_name

    Calling super().__init__() is optional — TaskMod keeps no instance state of its own; _prefix is a class-attribute default that :meth:with_prefix shadows on assignment. So your subclass __init__ can just store its parameters and return.

    For per-field CLI --help text, wrap the annotation in typing.Annotated[T, "help text"]. The string metadata is forwarded to argparse; the inner type is what drives validation::

    difficulty: Annotated[Literal["easy", "hard"], "Task difficulty preset."] = "easy"

    Implement :meth:apply to return the modified config; in-place mutation is fine because :meth:TaskRegistry.build calls cls() fresh on every build (and Pydantic deep-copies the config's BaseModel defaults per instance), so mutations do not leak across builds.

    Modifiers attach to a task by passing instances to @TASK_REGISTRY.register(modifiers=(...)) (a bare subclass is accepted and instantiated with no args). Modifiers are stored on the registry, not on the config class. A config registered without modifiers= inherits them from its nearest registered base class; modifiers=() opts out (see :meth:TaskRegistry.register).

    Per-instance scoping: chain :meth:with_prefix on a constructed instance so its CLI flags rename from <field> to <prefix>_<field>. The same subclass can be registered multiple times with distinct prefixes.

    Example

    ::

    from typing import Annotated, Literal

class RobotMod(TaskMod):
    def __init__(
        self,
        robot: Annotated[
            Literal["allegro", "shadow", "xhand1"],
            "Hand model to load into the scene.",
        ] = "xhand1",
        *,
        entity_name: str = "robot",  # keyword-only → not CLI-exposed
    ):
        self.robot = robot
        self.entity_name = entity_name

    def apply(self, config):
        setattr(config.scene_options, self.entity_name, ROBOT_REGISTRY.get(self.robot)())
        return config

@TASK_REGISTRY.register(modifiers=(
    RobotMod(robot="shadow", entity_name="robot_left").with_prefix("left"),
    RobotMod(entity_name="robot_right").with_prefix("right"),
))
class TwoArmTask(EdenRLConfig):

    def read_config_arg(path: str) -> dict:
    """Read the ``--config`` argument — :func:`read_config_file` plus the CLI's own redirect.

    The two flags each take what the other rejects, so each names the other when handed it:
    ``--override <a path>`` already points at ``--config``, and this is the mirror. The
    redirect fires on a *positive* identification — the value is something ``--override``
    itself would accept — not on a guess about the string, so a genuine path with the wrong
    extension (or unreadable contents) still gets the plain message.

    Parameters
    ----------
    path: str
        The ``--config`` value.

    Returns
    -------
    dict
        The file's contents, as :func:`read_config_file` returns them.

    Raises
    ------
    ValueError
        Naming ``--override`` when ``path`` is an inline mapping; otherwise unchanged from
        :func:`read_config_file`.
    """
    try:
        return read_config_file(path)
    except ValueError as e:
        if not _is_inline_mapping(path):
            raise  # a real path, just not one we can read — its own message is the right one
        raise ValueError(
            f"--config takes a .json/.yaml/.yml path, but {path!r} is an inline mapping. Pass it to --override instead."
        ) from e

    def explicit_task_mods(task_name: str, modifier_kwargs: dict[str, Any]) -> list[str]:
    """Return the ``--`` flags for task mods set away from their default.

    The comparison baseline is ``getattr(mod, field_name)`` — the mod *instance's* value,
    which is exactly what :func:`_add_mod_field_to_parser` uses as the argparse default (not
    the ``__init__`` signature default, which can differ when a task registers a mod with an
    overridden instance value). So a flag left unset — or passed at its default — reads as
    *not* explicitly set.

    Used to fail fast wherever mods collide with a config that is authoritative on its own
    (a saved ``--config``, a checkpoint's sibling ``config.json``) and would otherwise be
    silently ignored.

    Parameters
    ----------
    task_name: str
        The registered task whose mods to compare against.
    modifier_kwargs: dict[str, Any]
        Flag-keyed mod values, as produced by :func:`_collect_modifier_kwargs`.

    Returns
    -------
    list[str]
        The dash-prefixed flag names that differ from their default.
    """
    if not task_name or task_name not in TASK_REGISTRY:
        return []
    changed = []
    for mod in TASK_REGISTRY.get_modifiers(task_name):
        for field_name, _annotation, _default, _help in cli_fields(type(mod)):
            flag = mod.flag_for(field_name)
            if flag in modifier_kwargs and modifier_kwargs[flag] != getattr(mod, field_name):
                changed.append(f"--{flag}")
    return changed
