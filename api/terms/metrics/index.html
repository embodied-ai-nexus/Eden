    Metric Terms¶

    Metric terms grade a rollout: each reports a value per env with a direction (MetricDirection, higher or lower is better), a mode (MetricMode) and a success_threshold, and the metric manager combines the ones that contribute to success. Rewards and terminations can read a metric instead of recomputing it. The metrics and recorders guide explains them; the term catalog lists every parameter.

    from eden import metrics
from eden.constants import MetricDirection, MetricMode

    The base classes MetricTerm and MetricTermFuncWrapper are on the managers page.

def object_a_is_on_b(
    env: EnvBase,
    *,
    entity_a_name: str,
    entity_b_name: str,
    xy_threshold: float = 0.03,
    height_threshold: float = 0.04,
    height_diff: float = 0.0,
):
    """Check if object_a is on object_b by comparing height and xy distance of object origins.

    This function can be used as a termination term or a metric term.

    Parameters
    ----------
    env: EnvBase
        The environment instance.
    entity_a_name: str
        The name of the entity_a.
    entity_b_name: str
        The name of the entity_b.
    xy_threshold: float
        The threshold for the xy distance.
    height_threshold: float
        The threshold for the height distance.
    height_diff: float
        The expected height difference.

    Returns
    -------
    success: torch.Tensor
        A boolean tensor indicating if the object_a is on the object_b.
    """
    object_a: Entity = env.get_entity(entity_a_name)
    object_b: Entity = env.get_entity(entity_b_name)

    pos_diff = object_a.get_pos() - object_b.get_pos()
    height_dist = torch.linalg.vector_norm(pos_diff[:, 2:], dim=1)
    xy_dist = torch.linalg.vector_norm(pos_diff[:, :2], dim=1)

    success = torch.logical_and(
        pos_diff[:, 2] > 0.0,  # object_a is above object_b
        torch.logical_and(xy_dist < xy_threshold, (height_dist - height_diff) < height_threshold),
    )
    return success

def object_a_within_b_xy(
    env: EnvBase,
    *,
    entity_a_name: str,
    entity_b_name: str,
    min_fraction: float = 0.8,
) -> torch.Tensor:
    """*min_fraction* of entity_a's footprint lies over entity_b's.

    Both footprints are the xy extent of the entity's world-frame bounding box (its collision geometry),
    so a cup over a cutting board, or a scissor dropped into a container, passes whatever its height:
    pair it with a height or contact check to require that it rests there. PolaRiS's ``is_within_xy``
    rubric checker (``eden.tasks.polaris``) is this test.

    Parameters
    ----------
    env: EnvBase
        The environment instance.
    entity_a_name: str
        The entity that must lie over the other.
    entity_b_name: str
        The entity it must lie over.
    min_fraction: float
        Smallest share of entity_a's footprint area that must overlap entity_b's, in ``(0, 1]``.
    """
    a = env.get_entity(entity_a_name).get_AABB()[..., :2]  # (num_envs, 2 [low, high], 2)
    b = env.get_entity(entity_b_name).get_AABB()[..., :2]
    overlap = (torch.minimum(a[:, 1], b[:, 1]) - torch.maximum(a[:, 0], b[:, 0])).clamp(min=0.0).prod(dim=1)
    area = (a[:, 1] - a[:, 0]).prod(dim=1)
    return overlap >= min_fraction * area

    dist = torch.linalg.vector_norm(entity.get_pos() - target_pos, dim=1)
    return dist < threshold

    object_above_height ¶

    object_above_height(env: EnvBase, *, entity_name: str, height: float, relative: bool = False) -> Tensor

    Check if an entity's root z-position is above height.

    Returns a per-env bool tensor (True if above).

    PARAMETER DESCRIPTION
    env

    The environment instance.

    TYPE: EnvBase

    entity_name

    Name of the entity.

    TYPE: str

    height

    Minimum height threshold.

    TYPE: float

    relative

    Frame height is measured in. False (the default) reads the solver frame, which Genesis re-origins onto a rigid body's inertial frame; True reads the frame the asset was authored in. Pass True whenever height was reasoned from the asset's own geometry: the two frames differ by the body's centre-of-mass offset, and because that offset is expressed in the body frame, the reported z swings by it as the body tilts.

    Only a rigid entity has the second frame, so True against a deformable raises from :meth:~eden.entities.base.Entity._reject_relative. Unlike the observation terms, which the manager warm-computes while sizing their buffers, this is a metric term and so first runs on step 1 rather than at env construction.

    TYPE: bool DEFAULT: False

    SequentialMetricTerm(env: EnvBase, options: MetricTermOptions)

    Metric that tracks ordered sub-goals (phases).

    Each phase is a predicate that must be satisfied before the next phase is evaluated. The metric value is the fraction of phases completed (completed / total). With the default success_threshold=1.0 and direction="hib", the episode is considered successful once all phases are complete.

    Phase advancement is per-environment -- different parallel environments can be in different phases.

    PARAMETER DESCRIPTION
    phases

    Ordered list of phase definitions. Each PhaseOptions must contain one of:

    • term -- a MetricTermOptions from .configure().
    • ref -- a string referencing a sibling term in MetricManagerOptions.

    • threshold -- (default 1.0) minimum value returned by the phase predicate that counts as "passed".
    • hold -- if True, reuse the phase's own term as hold predicate.
    • hold_term -- a separate MetricTermOptions for the hold predicate (overrides hold).
    • hold_until -- the last phase (name or index, inclusive) through which the hold stays active; default: until the task completes.
    • hold_guard -- name of another held phase; this hold is enforced only where that phase's hold is also failing.

    def resolve_refs(self, manager_options) -> None:
    """Resolve ``ref`` phases by looking up sibling MetricTermOptions.

    Called by ``MetricManager._prepare_terms()`` in a second pass after
    all terms have been built.
    """
    for i, phase in enumerate(self.phases):
        if phase.ref is not None and self._resolved_phases[i][0] is None:
            sibling_options = getattr(manager_options, phase.ref, None)
            if sibling_options is None:

            term_instance = self._instantiate_term(sibling_options, f"phases[{i}].ref ('{phase.ref}')")

    # Now resolve hold conditions (requires all phases to be resolved)
    self._resolve_all_holds()

    def build(self):
    for i, phase in enumerate(self.phases):
        if not isinstance(phase, PhaseOptions):
            raise TypeError(
                f"Phase {i} must be a PhaseOptions instance, got {type(phase).__name__}. "
                f"Use PhaseOptions(term=MyTerm.configure(...), ...) instead of dict."
            )

        if phase.term is not None and phase.ref is not None:
            raise ValueError(f"Phase {i}: 'term' and 'ref' are mutually exclusive")
        if phase.term is None and phase.ref is None:
            raise ValueError(f"Phase {i}: must specify either 'term' or 'ref'")

        # Derive phase name
        name = phase.name
        if not name:
            if phase.term is not None:
                name = phase.term.name
            elif phase.ref is not None:
                name = phase.ref
            else:
                name = f"phase_{i}"
        self._phase_names.append(name)
        self._phase_name_to_idx[name] = i

        # Instantiate term (or defer if ref)
        if phase.term is not None:
            term_instance = self._instantiate_term(phase.term, f"phases[{i}].term")
            self._resolved_phases.append((term_instance, phase.threshold))
        else:
            # ref — will be resolved in resolve_refs()
            self._resolved_phases.append((None, phase.threshold))

    # If no refs, resolve holds now; otherwise MetricManager will call resolve_refs()
    has_refs = any(p.ref is not None for p in self.phases)
    if not has_refs:
        self._resolve_all_holds()

    def compute(self) -> torch.Tensor:
    num_phases = len(self._resolved_phases)
    if num_phases == 0:
        return torch.ones(self._env.num_envs, device=self.device)

    # ------------------------------------------------------------------
    # Hold-condition check: revert to earliest failing hold
    # ------------------------------------------------------------------
    if self._has_any_holds:
        # Track the earliest failing phase per env (num_phases = "no failure")
        revert_target = torch.full((self._env.num_envs,), num_phases, dtype=torch.long, device=self.device)

        # Pre-compute hold values so guards can reference them.
        # hold_failed[p] is a per-env bool tensor (True = hold failing).
        hold_failed: list[torch.Tensor | None] = [None] * num_phases
        for p, hold in enumerate(self._hold_conditions):
            if hold is None:
                continue
            candidates = (self._phase_buf > p) & (self._phase_buf < hold.until_idx)
            if not candidates.any():
                continue
            h_value = hold.term.compute()
            h_failed = h_value < hold.threshold
            hold_failed[p] = candidates & h_failed

            if en.logger.level <= logging.DEBUG:
                for e in range(self._env.num_envs):
                    if candidates[e]:
                        en.logger.debug(
                            f"  [env {e}] hold check phase {p} ({self._phase_names[p]}): "
                            f"value={h_value[e].item():.4f}  threshold={hold.threshold}  "
                            f"failed={bool(h_failed[e])}"
                        )

        # Apply hold failures, respecting guards.
        for p, hold in enumerate(self._hold_conditions):
            if hold is None or hold_failed[p] is None:
                continue
            failing = hold_failed[p]
            if not failing.any():
                continue

            # If this hold has a guard, suppress it for envs where the
            # guard's hold is still passing (i.e. guard not failing).
            if hold.guard_idx is not None:
                guard_failing = hold_failed[hold.guard_idx]
                if guard_failing is None:
                    # Guard hold wasn't even evaluated (no envs in window) → guard is passing → suppress all
                    if en.logger.level <= logging.DEBUG:
                        en.logger.debug(
                            f"  hold on phase {p} ({self._phase_names[p]}) fully suppressed "
                            f"by guard '{self._phase_names[hold.guard_idx]}' (guard passing)"
                        )
                    continue
                # Only enforce this hold where the guard is also failing
                failing = failing & guard_failing
                if not failing.any():
                    if en.logger.level <= logging.DEBUG:
                        en.logger.debug(
                            f"  hold on phase {p} ({self._phase_names[p]}) suppressed "
                            f"by guard '{self._phase_names[hold.guard_idx]}' (guard passing for all envs)"
                        )
                    continue

            revert_target = torch.where(failing & (p < revert_target), self._phase_tensors[p], revert_target)

        # Apply reverts
        needs_revert = revert_target < num_phases
        if needs_revert.any():
            if en.logger.level <= logging.INFO:
                for e in range(self._env.num_envs):
                    if needs_revert[e]:
                        old_p = self._phase_buf[e].item()
                        new_p = revert_target[e].item()
                        en.logger.info(
                            f"[env {e}] HOLD FAILED: reverting from phase {old_p} ({self._phase_names[old_p]}) "
                            f"→ phase {new_p} ({self._phase_names[new_p]})"
                        )
            self._phase_buf = torch.where(needs_revert, revert_target, self._phase_buf)

    # ------------------------------------------------------------------
    # Phase advancement
    # ------------------------------------------------------------------
    prev_phases = self._phase_buf.clone()

    # When allow_skip is False, use a snapshot so each env advances at
    # most one phase per compute().  When True, read _phase_buf directly
    # so envs that just advanced are re-evaluated for the next phase.
    phase_view = self._phase_buf if self.allow_skip else prev_phases

    for p, (term_instance, threshold) in enumerate(self._resolved_phases):
        mask = phase_view == p
        if not mask.any():
            continue
        value = term_instance.compute()
        passed = value >= threshold
        advance = mask & passed
        self._phase_buf[advance] = p + 1

        if en.logger.level <= logging.DEBUG:
            for e in range(self._env.num_envs):
                if mask[e]:
                    en.logger.debug(
                        f"  [env {e}] phase {p} ({self._phase_names[p]}): "
                        f"value={value[e].item():.4f}  threshold={threshold}  "
                        f"passed={bool(passed[e])}"
                    )

    # Clamp to [0, num_phases]
    self._phase_buf.clamp_(max=num_phases)

    if en.logger.level <= logging.INFO:
        for e in range(self._env.num_envs):
            if self._phase_buf[e] != prev_phases[e]:
                old_p = prev_phases[e].item()
                new_p = self._phase_buf[e].item()
                old_name = self._phase_names[old_p]
                if new_p >= num_phases:
                    en.logger.info(
                        f"[env {e}] Phase {old_p} ({old_name}) DONE → task COMPLETE ({new_p}/{num_phases})"
                    )
                else:
                    en.logger.info(
                        f"[env {e}] Phase {old_p} ({old_name}) DONE → now at phase {new_p} ({self._phase_names[new_p]})  [{new_p}/{num_phases}]"
                    )

    return self._phase_buf.float() / num_phases

    RubricMetricTerm ¶

    RubricMetricTerm(env: EnvBase, options: MetricTermOptions)

    Bases: MetricTerm

    Fraction of rubric criteria met, each after its dependencies.

    A criterion is reached the first step its predicate succeeds — by the term's own is_success, so its direction and success_threshold decide (entity_distance with success_threshold=0.05 is met within 5 cm) — while every criterion in its after list has already been reached, and stays reached until the env resets: a cube let go over the tray counts even if it bounces off later. The value is the fraction of criteria reached, so with the default success_threshold=1.0 the task succeeds once all of them are. Unlike :class:SequentialMetricTerm, criteria with no dependency between them are reached in any order (both cubes' "reach" before either "lift", or one cube's whole chain first) — the progress score of PolaRiS's Rubric, which :mod:eden.tasks.polaris reproduces.

    Criteria are evaluated in list order every step, so after names earlier criteria only, and a criterion can be reached in the same step as the ones it depends on.

    PARAMETER DESCRIPTION
    criteria

    The criteria, each a named predicate (term) and the earlier criteria it needs (after).

    TYPE: list[CriterionOptions]

    Example

    ::

    RubricMetricTerm.configure(
    criteria=[
        CriterionOptions(name="reach", term=EeNearEntity.configure(robot_name="gripper", ...)),
        CriterionOptions(name="lift", term=object_above_height.configure(...), after=["reach"]),
        CriterionOptions(name="in_bowl", term=object_a_within_b_xy.configure(...), after=["lift"]),
    ]
)

    def build(self):
    if not self.criteria:
        raise ValueError(
            "RubricMetricTerm needs at least one criterion: pass criteria=[CriterionOptions(...), ...]."
        )
    index: dict[str, int] = {}
    for i, criterion in enumerate(self.criteria):
        if not isinstance(criterion, CriterionOptions):
            raise TypeError(
                f"criteria[{i}] must be a CriterionOptions, got {type(criterion).__name__}; write "
                f"CriterionOptions(name=..., term=MyTerm.configure(...)) rather than a dict."
            )
        if criterion.name in index:
            raise ValueError(
                f"RubricMetricTerm criterion name '{criterion.name}' is used twice; names must be unique."
            )
        unknown = [name for name in criterion.after if name not in index]
        if unknown:
            raise ValueError(
                f"RubricMetricTerm criterion '{criterion.name}' lists after={unknown}, which are not earlier "
                f"criteria; earlier ones: {list(index) or 'none'}. Move the criterion below the ones it depends on."
            )
        term = build_term(
            METRIC_TERM_REGISTRY, self._env, criterion.term, label=f"{type(self).__name__}.criteria[{i}]"
        )
        if term.is_cumulative:
            raise ValueError(
                f"RubricMetricTerm criterion '{criterion.name}' has is_cumulative=True, but a criterion is "
                f"judged on each step's value, never on a sum over the episode; configure its term with "
                f"is_cumulative=False."
            )
        self._after.append([index[name] for name in criterion.after])
        self._terms.append(term)
        index[criterion.name] = i

    def compute(self) -> torch.Tensor:
    for i, (term, after) in enumerate(zip(self._terms, self._after)):
        met = term.is_success(term.compute())
        if after:
            met &= self._reached[:, after].all(dim=1)
        self._reached[:, i] |= met
    return self._reached.float().mean(dim=1)
