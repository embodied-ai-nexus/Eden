    Recorder Terms¶

    Recorder terms capture rollout data (actions, observations, RL transitions, entity states, sensor readings, system-identification signals) for the recorder manager to write to an episode file. The metrics and recorders guide and the datasets and recording guide explain the file layout; the term catalog lists every parameter.

    from eden import recorders

    The base class RecorderTerm is on the managers page.

    ObservationRecorder(env: EnvBase, options: _TermOptionsT)

    Bases: RecorderTerm

    Record observation groups at the start of each step, as obs/<group>.

    The start of a step is when the policy — or a scripted demonstrator — reads the observations and picks the action :class:~eden.managers.terms.recorders.actions.ActionRecorder records for that same step, so row t of obs and action form one (o_t, a_t) pair: the shape a behavior-cloning dataset needs.

    def record_pre_step(self) -> dict[str, torch.Tensor | dict]:
    obs = self._env.obs_buf
    if obs is None:
        raise RuntimeError(
            "ObservationRecorder recorded a step before the env computed any observations; reset the env "
            "(env.reset()) before stepping it."
        )
    groups = self.groups if self.groups is not None else list(obs)
    return {"obs": {group: _clone(obs[group]) for group in groups}}

    def record_post_step(self) -> dict[str, torch.Tensor]:
    out: dict[str, torch.Tensor] = {}
    if self.include_reward and hasattr(self._env, "reward_buf"):
        out["reward"] = self._env.reward_buf.unsqueeze(-1)
    if self.include_done and hasattr(self._env, "reset_buf"):
        out["done"] = self._env.reset_buf.to(dtype=torch.bool).unsqueeze(-1)
    if self.include_terminated and hasattr(self._env, "reset_terminated"):
        out["terminated"] = self._env.reset_terminated.to(dtype=torch.bool).unsqueeze(-1)
    if self.include_timeout and hasattr(self._env, "reset_timeouts"):
        out["timeout"] = self._env.reset_timeouts.to(dtype=torch.bool).unsqueeze(-1)
    if self.include_timestamp and hasattr(self._env, "episode_length_buf") and hasattr(self._env, "dt"):
        out["timestamp"] = self._env.episode_length_buf.to(dtype=torch.float32).unsqueeze(-1) * float(self._env.dt)
    return out

    def record_post_reset(self, envs_idx=None) -> dict[str, torch.Tensor | dict]:
    if not self._env.entities:
        return {}

    entities_state: dict[str, dict[str, torch.Tensor]] = {}
    for entity_name, entity in self._env.entities.items():
        entity_state: dict[str, torch.Tensor] = {}

        if self.include_qpos:
            entity_state["qpos"] = entity.get_qpos(qs_idx_local=entity.qs_idx_local, envs_idx=envs_idx).clone()
        if self.include_dofs_vel:
            entity_state["dofs_vel"] = entity.get_dofs_vel(envs_idx=envs_idx).clone()
        if self.include_pos_quat:
            entity_state["pos"] = entity.get_pos(envs_idx=envs_idx).clone()
            entity_state["quat"] = entity.get_quat(envs_idx=envs_idx).clone()

        if entity_state:
            entities_state[entity_name] = entity_state

    if not entities_state:
        return {}

    return {"initial_state": {"entities": entities_state}}

    StateRecorder(env: EnvBase, options: _TermOptionsT)

    SysIDRecorder(env: EnvBase, options: _TermOptionsT)

    Bases: RecorderTerm

    Record the proprioceptive trace needed to rebuild a sysid Trajectory.

    Captures the signals under the sysid/ namespace so that eden.extensions.sysid.Trajectory.from_recorder_episode can reload an episode without any extra metadata.

    PARAMETER DESCRIPTION
    entity_name

    Entity to record. Must exist in env.entities.
    include_torque

    Record get_dofs_control_force() under sysid/dofs_control_force.
    include_base

    Record the base pose (pos/quat) at reset and the base quat / body-frame ang_vel each step. Set False for fixed-base arms.

    def record_post_reset(self, envs_idx=None) -> dict[str, torch.Tensor | dict]:
    entity = self.resolve_entity(self.entity_name)
    initial_state: dict[str, torch.Tensor] = {
        "qpos": entity.get_qpos(qs_idx_local=entity.qs_idx_local, envs_idx=envs_idx).clone(),

    def record_post_step(self) -> dict[str, torch.Tensor | dict]:
    entity = self.resolve_entity(self.entity_name)
    out: dict[str, torch.Tensor] = {
        "dofs_pos": entity.get_dofs_pos().clone(),
        "dofs_vel": entity.get_dofs_vel().clone(),
    }
    if self.include_torque:
        out["dofs_control_force"] = entity.get_dofs_control_force().clone()
    if self.include_base:
        out["base_quat"] = entity.get_quat().clone()
        out["base_ang_vel"] = entity.get_ang(frame="body").clone()
    return {"sysid": out}
