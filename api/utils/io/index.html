    I/O Utilities¶

    Reading and writing episode data, converting recordings to LeRobot datasets, image conversion, and reading body-model pickles without chumpy. The datasets and recording guide explains the recording workflow.

    File handlers¶

    The recorder manager writes episodes through a file handler, chosen by name from FILE_HANDLER_REGISTRY (HDF5 or NPZ). EpisodeData is the in-memory container of one recorded trajectory.

    Base class and registry for dataset file handlers.

    def __init__(self, options: T) -> None:
    self._uid = gs.UID()
    self._options: T = options

    mirror_options_onto(self, self._options)

    # Deferred, not module-scope — see the NOTE above this module's imports.
    from eden.options._metadata import OPTION_METADATA_KEYS

    # check if there are unused parameters in the options. ``_extra_option_params`` are legitimate
    # options-only fields (not mirrored as attrs), so they are expected to be "unused" here.
    unused_params = set(self._options.keys()) - set(self.get_parameter_names())
    tolerated = {"name", "class_name"} | set(OPTION_METADATA_KEYS) | self._extra_option_params
    # ``params`` is a declared framework field on function-capable term options. An *empty* one is
    # legitimately unused when the same options class configures a class-based term; a non-empty
    # ``params`` on a class-based term would be silently ignored, so it stays invalid.
    if not getattr(self._options, "params", None):
        tolerated.add("params")
    invalid_params = unused_params - tolerated
    if invalid_params:
        # Name the term, not only the class: for a function term the class is the shared
        # ``*FuncWrapper``, which cannot tell the author which of a dozen terms is wrong.
        # Same wording as the construction-time check (``ManagerTermOptions._validate_param_names``).
        term_name = getattr(self._options, "name", None)
        subject = f"term '{term_name}' ({type(self).__name__})" if term_name else type(self).__name__
        raise ValueError(
            invalid_parameters_message(subject, invalid_params, self.get_configure_specs(), self._retired_params)
        )

    def add(self, key: str, value: torch.Tensor | dict):
    """Add a key-value pair to the dataset.

    The key can be nested by using the "/" character.
    For example: "obs/joint_pos".

    Parameters
    ----------
    key : str
        The key name.
    value : torch.Tensor or dict
        The corresponding value, either a tensor or a (possibly nested) dict.
    """
    if isinstance(value, dict):
        for sub_key, sub_value in value.items():
            self.add(f"{key}/{sub_key}", sub_value)
        return

    self._append_owned(key, value.clone())

    def get(self, key: str, index: int | None = None, sequential: bool = False):
    """General getter for any key in the episode data.

    Parameters
    ----------
    key : str
        Hierarchical key, e.g. ``'states/obs'``, ``'actions'`` or ``'rewards'``.
    index : int, optional
        Specific timestep to retrieve.
    sequential : bool, optional
        If True, use and increment a per-key pointer.

    Returns
    -------
    torch.Tensor or dict or None
        The tensor (or dict of tensors) at the requested timestep, or None if
        the key is missing or the index is out of range.

    Examples
    --------
    >>> # Random access
    >>> state5 = episode.get("states/obs/joint_pos", index=5)
    >>> action3 = episode.get("actions", index=3)
    >>> # Sequential access
    >>> next_state = episode.get("states/obs", sequential=True)
    >>> next_action = episode.get("actions", sequential=True)
    """
    data_pointer = self._resolve_key(key)
    if data_pointer is None:
        return None

    # Determine the index
    if sequential:
        idx = self._next_index_dict.get(key, 0)
        self._next_index_dict[key] = idx + 1
    elif index is not None:
        idx = index
    else:
        idx = 0

    # Access the data
    if isinstance(data_pointer, (list, torch.Tensor)):
        if idx >= len(data_pointer):
            return None
        return data_pointer[idx] if isinstance(data_pointer, list) else data_pointer[idx, ...]
    elif isinstance(data_pointer, dict):
        return {subkey: self.get(f"{key}/{subkey}", index=idx) for subkey in data_pointer}
    return None

    def size(self, key: str):
    """Return the size of the data at the given key.

    Parameters
    ----------
    key : str
        Hierarchical key, e.g. ``'states/obs'``, ``'actions'`` or ``'rewards'``.

    Returns
    -------
    int or dict or None
        Size of the stored value, a dict of sizes for nested keys, or None if
        the key is missing.
    """
    data_pointer = self._resolve_key(key)
    if data_pointer is None:
        return None

    if isinstance(data_pointer, (list, torch.Tensor)):
        return len(data_pointer)
    elif isinstance(data_pointer, dict):
        return {subkey: self.size(f"{key}/{subkey}") for subkey in data_pointer}
    return 0

    def __init__(self):
    """Initialize the HDF5 dataset file handler."""
    self._hdf5_file_stream = None
    self._hdf5_data_group = None
    # _next_demo_id is the next writable demo index; _demo_count is the actual
    # episode count. They diverge when a resumed dataset has gaps in numbering
    # (e.g. demo_0, demo_2 after a crash → next_id=3, count=2), and keeping both means
    # the next write never collides and get_num_episodes() never over-reports.
    self._next_demo_id = 0
    self._demo_count = 0

    def open(self, file_path: str, mode: str = "r", env_cfg: dict | None = None):
    """Open an existing dataset file."""
    if self._hdf5_file_stream is not None:
        raise RuntimeError("HDF5 dataset file stream is already in use")
    file_path = self.resolve_path(file_path)
    self._hdf5_file_stream = h5py.File(file_path, mode)

    if "data" not in self._hdf5_file_stream:
        raise RuntimeError("Invalid dataset file: missing 'data' group")

    self._hdf5_data_group = self._hdf5_file_stream["data"]
    existing_indices = [
        int(m.group(1)) for m in (_DEMO_NAME_RE.fullmatch(name) for name in self._hdf5_data_group.keys()) if m
    ]
    self._next_demo_id = max(existing_indices) + 1 if existing_indices else 0
    self._demo_count = len(existing_indices)

    if env_cfg is not None and "env_cfg" in self._hdf5_data_group.attrs:
        try:
            stored_cfg = json.loads(self._hdf5_data_group.attrs["env_cfg"])
        except (ValueError, json.JSONDecodeError):
            stored_cfg = None
        if stored_cfg is not None and stored_cfg != env_cfg:
            warnings.warn(
                f"Resumed HDF5 dataset '{file_path}' was recorded with a different env_cfg; "
                "mixing episodes from different configs may break downstream consumers.",
                stacklevel=2,
            )

    def load_episode(self, episode_name: str, device: str = "cpu") -> EpisodeData | None:
    self._raise_if_not_initialized()

    attrs_key = f"{episode_name}{_ATTRS_SUFFIX}"

    has_any = False
    for key, value in self._arrays.items():
        if not key.startswith(prefix):

        has_any = True
        # Skip the attrs key
        if key == attrs_key:

        sub_key = key[len(prefix) :]

                if attr == "success":
                    value = bool(value)
                elif attr in ("seed", "env_id"):
                    value = int(value)
                setattr(episode, attr, value)

    def flush(self):
    # NPZ has no incremental write format, so flush rewrites the full archive,
    # atomically (tmp file + rename) so a kill mid-write cannot leave a
    # partially-written dataset. Its cost is O(total dataset size): for large
    # datasets prefer close() over a flush() per episode.
    self._raise_if_not_initialized()
    if not self._writable or not self._dirty:
        return
    tmp_path = f"{self._file_path}.tmp.npz"
    with open(tmp_path, "wb") as f:
        np.savez_compressed(f, **self._arrays)
    os.replace(tmp_path, self._file_path)
    self._dirty = False

    hdf5_path: Path,

    """Convert an HDF5 recording into a LeRobot v3 dataset on disk."""
    # Before --force deletes anything: a missing or outdated lerobot must not cost the user their old dataset.
    LeRobotDataset = _import_lerobot_dataset()

    if not hdf5_path.is_file():
        raise FileNotFoundError(f"Input HDF5 file not found: {hdf5_path}")

    if output_dir.exists():
        if not force:
            raise FileExistsError(f"Output directory already exists: {output_dir}")
        if output_dir.is_symlink() or not output_dir.is_dir():
            raise ValueError(f"Refusing to --force overwrite non-directory path: {output_dir}")
        if any(output_dir.iterdir()) and not _looks_like_lerobot_dataset(output_dir):
            raise ValueError(
                f"Refusing to --force overwrite {output_dir}: it is not empty and does not look "
                "like a LeRobot dataset (missing meta/info.json). Delete it manually if intended."
            )
        shutil.rmtree(output_dir)

    with h5py.File(hdf5_path, "r") as hdf5_file:
        demos = _list_demos(hdf5_file)
        features = _build_features(hdf5_file, demos)
        fps = _infer_fps(hdf5_file[f"data/{demos[0]}/timestamp"][:])

        dataset = LeRobotDataset.create(
            repo_id=repo_id,
            root=output_dir,
            fps=fps,
            robot_type=robot_type,
            features=features,
            use_videos=True,
            image_writer_threads=image_writer_threads,
        )

        for demo in demos:
            demo_group = hdf5_file[f"data/{demo}"]
            # Bulk-load each demo's arrays once instead of per-step h5 indexing.
            actions = np.asarray(demo_group["action"][:], dtype=np.float32)
            timestamps = _normalize_timestamps(demo_group["timestamp"][:])
            qpos = np.asarray(demo_group["state/qpos"][:], dtype=np.float32)
            dofs_vel = np.asarray(demo_group["state/dofs_vel"][:], dtype=np.float32)
            pos = np.asarray(demo_group["state/pos"][:], dtype=np.float32)
            quat = np.asarray(demo_group["state/quat"][:], dtype=np.float32)

            expected_lengths = {
                "action": actions.shape[0],
                "timestamp": timestamps.shape[0],
                "qpos": qpos.shape[0],
                "dofs_vel": dofs_vel.shape[0],
                "pos": pos.shape[0],
                "quat": quat.shape[0],
            }
            if len(set(expected_lengths.values())) != 1:
                raise ValueError(f"Length mismatch in {demo}: {expected_lengths}")

            states = np.concatenate([qpos, dofs_vel, pos, quat], axis=1)
            length = int(actions.shape[0])
            for t in range(length):
                dataset.add_frame(
                    {
                        "observation.state": states[t],
                        "action": actions[t],
                        "task": task,
                    }
                )

            dataset.save_episode()

        dataset.finalize()

    Eden camera terms (:func:eden.managers.terms.observations.camera.rgb_image) return (num_envs, H, W, C) uint8 tensors. Consumers (policy servers, recorders, ...) differ only in layout (HWC vs CHW) and in how they want the image resized, so those are the two axes exposed here:

    • :func:to_uint8_hwc / :func:to_chw_uint8 — layout, no resampling.
    • :func:resize_uint8 — plain resize (cv2 bilinear; Isaac-GR00T's SimplerEnv eval).
    • :func:resize_with_pad — aspect-preserving letterbox resize, a bit-exact port of openpi_client.image_tools.resize_with_pad (pinned by tests/utils/test_image.py) so OpenPI-family clients do not need openpi_client installed.

    def to_uint8_hwc(image: torch.Tensor | np.ndarray) -> np.ndarray:
    """``(1, H, W, C)`` or ``(H, W, C)`` -> ``(H, W, 3)`` ``uint8`` numpy.

    A 4-channel (RGBA, as the madrona/Nyx renderers return) input has its alpha channel
    dropped. Any other channel count, dtype, or a batch axis larger than one raises: a
    float image or a batched render reaching a server is a wiring error, not something
    to rescale or slice quietly.
    """
    arr = image.detach().cpu().numpy() if isinstance(image, torch.Tensor) else np.asarray(image)
    if arr.ndim == 4:
        if arr.shape[0] != 1:
            raise ValueError(f"Expected a single-env image with leading axis 1, got shape {arr.shape}.")
        arr = arr[0]
    if arr.ndim != 3:
        raise ValueError(f"Expected an HxWxC image (optionally with a leading batch axis), got shape {arr.shape}.")
    if arr.shape[-1] == 4:
        arr = arr[..., :3]
    if arr.shape[-1] != 3:
        raise ValueError(f"Expected 3 (or 4 with alpha) channels, got shape {arr.shape}.")
    if arr.dtype != np.uint8:
        raise TypeError(
            f"Expected a uint8 image, got dtype {arr.dtype}. Eden's rgb_image term returns uint8; "
            "convert explicitly at the source if a renderer produces floats."
        )
    return arr

    def resize_uint8(image_np: np.ndarray, image_size: tuple[int, int]) -> np.ndarray:
    """Resize an ``(H, W, 3)`` ``uint8`` image to ``image_size=(h, w)`` with cv2 bilinear filtering.

    No-op when the size already matches. Matches ``gr00t/eval/sim/SimplerEnv/simpler_env.py``.
    """
    import cv2

    h, w = image_size
    if image_np.shape[:2] != (h, w):
        image_np = cv2.resize(image_np, (w, h))
    if image_np.dtype != np.uint8:
        image_np = image_np.astype(np.uint8)
    return image_np

    Body-model pickles¶

    MANO, SMPL and SMPL-H .pkl files pickle chumpy arrays; these helpers read them without installing chumpy.

    Read the body-model pickles that hold chumpy arrays (MANO, SMPL, SMPL-H) without chumpy.

    Those releases pickle shapedirs as a chumpy.ch.Ch (SMPL) or a chumpy.reordering.Select over one (MANO, SMPL-H). chumpy itself is unmaintained, its PyPI release does not build on a current numpy, and a published wheel cannot depend on the git commit that does, so Eden reads the two classes itself: :class:ChumpyFreeUnpickler for Eden's own loader (:func:eden.datasets.utils.body_model.body_model_arrays), and :func:install_chumpy_pickle_modules for a third-party loader that calls a plain pickle.load (smplx, in the SOMA retargeter's worker process).

    numpy-only on purpose: the SOMA worker imports it and must stay free of Genesis.

    def find_class(self, module: str, name: str):
    if module == "chumpy" or module.startswith("chumpy."):
        cls = _CHUMPY_CLASSES.get((module, name))
        if cls is None:
            known = ", ".join(f"{m}.{n}" for m, n in _CHUMPY_CLASSES)
            raise pickle.UnpicklingError(
                f"This body-model pickle holds a {module}.{name}, a chumpy class Eden does not read (it reads "
                f"{known}: every chumpy class in the MANO, SMPL and SMPL-H releases). Use the model's .npz "
                f"instead (put it next to the .pkl), or convert the .pkl to an .npz with the same keys. "
                f"Installing chumpy does not help: Eden never imports it."
            )
        return cls
    return super().find_class(module, name)

    def install_chumpy_pickle_modules() -> None:
    """Make ``chumpy.ch.Ch`` / ``chumpy.reordering.Select`` resolve to Eden's stand-ins, process-wide.

    For a third-party loader that reads a model ``.pkl`` with a plain ``pickle.load`` (smplx, under the SOMA
    retargeter), which a custom unpickler cannot reach. It registers ``chumpy``, ``chumpy.ch`` and
    ``chumpy.reordering`` in ``sys.modules``, shadowing an installed chumpy for the rest of the process, so only a
    process that exists to run such a loader calls it (the SOMA worker). The stand-ins read every MANO, SMPL and
    SMPL-H release bit-identically to chumpy. A chumpy the process has already imported is left in place; a
    ``sys.modules["chumpy"] = None`` (an import blocked on purpose) is not an import, and is replaced.
    """
    if sys.modules.get("chumpy") is not None:
        return
    modules = {name: types.ModuleType(name) for name in ("chumpy", "chumpy.ch", "chumpy.reordering")}
    for (module, name), cls in _CHUMPY_CLASSES.items():
        setattr(modules[module], name, cls)
    modules["chumpy"].ch = modules["chumpy.ch"]
    modules["chumpy"].reordering = modules["chumpy.reordering"]
    sys.modules.update(modules)
